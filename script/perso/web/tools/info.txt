 SELECT
    m.release_year,
    COUNT(*) as nombre_de_streams
FROM scrobbles s
JOIN metadata_ref m ON s.track_id = m.track_id
WHERE m.release_year IS NOT NULL
GROUP BY m.release_year
ORDER BY nombre_de_streams DESC;
-- 1. Nettoyage des titres avec " (Feat. ...)"
UPDATE metadata_ref
SET track = "'" || TRIM(SUBSTR(track, 2, INSTR(track, ' (Feat.') - 2)) || "'"
WHERE track LIKE '% (Feat.%';

-- 2. Nettoyage des titres avec " (feat. ...)" (au cas où la casse varie)
UPDATE metadata_ref
SET track = "'" || TRIM(SUBSTR(track, 2, INSTR(track, ' (feat.') - 2)) || "'"
WHERE track LIKE '% (feat.%';
